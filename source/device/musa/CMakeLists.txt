UNSET (_DEV_MUSA_HEADER_PATH)
UNSET (_MUSA_BASE_SOURCE)
UNSET (_MUSA_OPS_SOURCE)
UNSET (_DEV_MUSA_DEVICE_SOURCE)
UNSET (_DEV_MUSA_COMPILER_DEFINES)
UNSET (_DEV_MUSA_COMPILER_OPTIONS)
UNSET (_DEV_MUSA_LINKER_OPTIONS)
UNSET (_DEV_MUSA_LINK_LIBRARIES)

SET(_MUSA_ROOT ${CMAKE_SOURCE_DIR}/source/device/musa)

LIST (APPEND _DEV_MUSA_HEADER_PATH      ${_MUSA_ROOT})
LIST (APPEND _DEV_MUSA_HEADER_PATH      ${MUSA_INCLUDE_DIRS})

LIST (APPEND _DEV_MUSA_LINK_PATH        "${MUSA_TOOLKIT_ROOT}/lib")

AUX_SOURCE_DIRECTORY("${_MUSA_ROOT}"    _MUSA_BASE_SOURCE)
LIST (APPEND _DEV_MUSA_DEVICE_SOURCE    ${_MUSA_BASE_SOURCE})

FILE (GLOB _MUSA_MU_SOURCES "${_MUSA_ROOT}/op/*.mu")

LIST (APPEND _DEV_MUSA_LINK_LIBRARIES   ${MUSA_RUNTIME_LIBRARY})
LIST (APPEND _DEV_MUSA_LINK_LIBRARIES   ${MUBLAS_LIBRARY})
LIST (APPEND _DEV_MUSA_LINK_LIBRARIES   ${MUDNN_LIBRARY})

SET (TENGINE_MUSA_HEADER_PATH       ${_DEV_MUSA_HEADER_PATH}        CACHE INTERNAL  "Tengine MUSA device header files searching path"   FORCE)
SET (TENGINE_MUSA_LINK_PATH         ${_DEV_MUSA_LINK_PATH}          CACHE INTERNAL  "Tengine MUSA device link libraries searching path" FORCE)
SET (TENGINE_MUSA_DEVICE_SOURCE     ${_DEV_MUSA_DEVICE_SOURCE}      CACHE INTERNAL  "Tengine MUSA device main source files"             FORCE)
SET (TENGINE_MUSA_MU_SOURCES        ${_MUSA_MU_SOURCES}             CACHE INTERNAL  "Tengine MUSA device .mu source files"              FORCE)
SET (TENGINE_MUSA_COMPILER_DEFINES  ${_DEV_MUSA_COMPILER_DEFINES}   CACHE INTERNAL  "Tengine MUSA about compiler defines"               FORCE)
SET (TENGINE_MUSA_COMPILER_OPTIONS  ${_DEV_MUSA_COMPILER_OPTIONS}   CACHE INTERNAL  "Tengine MUSA about compiler options"               FORCE)
SET (TENGINE_MUSA_LINKER_OPTIONS    ${_DEV_MUSA_LINKER_OPTIONS}     CACHE INTERNAL  "Tengine MUSA about linker options"                 FORCE)
SET (TENGINE_MUSA_LINK_LIBRARIES    ${_DEV_MUSA_LINK_LIBRARIES}     CACHE INTERNAL  "Tengine MUSA about link libraries"                 FORCE)
